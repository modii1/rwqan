✔️ 1) أضف اسم الشيت في SHEETS

ابحث عن:

const SHEETS = {


ثم أضف:

VERIFICATION_LOGS: "سجل التحقق",

✔️ 2) أضف الدوال داخل GoogleSheetsService
🔽 أضف هذه الدوال داخل الكلاس:
// ======================
// 📌 سجل التحقق (GS)
// ======================

async addVerificationLogToSheet(log: {
  date: string;
  propertyNumber: string;
  action: string;
  reason: string;
  admin: string;
}) {
  const row = [
    log.date,
    log.propertyNumber,
    log.action,
    log.reason,
    log.admin,
  ];

  await this.appendToSheet(SHEETS.VERIFICATION_LOGS, [row]);
}

async getVerificationLogsFromSheet() {
  const rows = await this.readSheet(SHEETS.VERIFICATION_LOGS);

  return rows.map((r) => ({
    date: r[0] || "",
    propertyNumber: r[1] || "",
    action: r[2] || "",
    reason: r[3] || "",
    admin: r[4] || "",
  }));
}

⭐ ثالثاً: تعديل storage.ts ليستخدم الشيت بدل الذاكرة

افتح:

server/storage.ts


واحذف أو تجاهل النظام القديم:

private verificationLogs = [];


واستبدل دواله بهذه النسخة:

✔️ 1) إضافة السجل
async addVerificationLog(log: {
  propertyNumber: string;
  action: "approved" | "rejected";
  reason?: string;
  date: string;
  admin?: string;
}) {
  await googleSheetsService.addVerificationLogToSheet({
    date: log.date,
    propertyNumber: log.propertyNumber,
    action: log.action,
    reason: log.reason || "",
    admin: log.admin || "Admin",
  });

  return log;
}

✔️ 2) جلب السجلات
async getVerificationLogs() {
  return googleSheetsService.getVerificationLogsFromSheet();
}

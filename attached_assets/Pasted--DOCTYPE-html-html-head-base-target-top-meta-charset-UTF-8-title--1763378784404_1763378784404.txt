      "مخيم","نظام قسمين رجال ونساء","ملحق","ممشى صحي"
    ];
    const container = document.getElementById("facilitiesGroup");
    facilitiesList.forEach((facility, idx) => {
      const id = "facility_" + idx;
      const label = document.createElement("label");
      label.htmlFor = id;
      label.innerHTML = `<input type="checkbox" id="${id}" name="facilities" value="${facility}" /> ${facility}`;
      container.appendChild(label);
    });

    function togglePriceField(checkbox){
      const field = document.querySelector(`input[name='${checkbox.value}']`);
      field.style.display = checkbox.checked ? "block" : "none";
      field.required = checkbox.checked;
      if(!checkbox.checked) field.value = ""; // ← حذف القيمة القديمة تمامًا
    }

    const propertyNumberInput = document.querySelector("input[name='propertyNumber']");
    const requestTypeSelect = document.querySelector("select[name='requestType']");
    const statusDiv = document.getElementById("status");
    const form = document.getElementById("propertyForm");

    let propertyTimeout;
propertyNumberInput.addEventListener("input", function(){
  if(propertyTimeout) clearTimeout(propertyTimeout);

  // نبحث فقط إذا طول الرقم 3+ أرقام
  const number = propertyNumberInput.value.trim();
  if(number.length < 4) {
    statusDiv.innerText = "أدخل 3 أرقام على الأقل للبحث...";
    statusDiv.className = "status";
    return;
  }

  propertyTimeout = setTimeout(()=>{ 
    loadPropertyData(); 
  }, 2000); // زيادة التأخير قليلاً لمنح المستخدم فرصة الإكمال
});


    requestTypeSelect.addEventListener("change", loadPropertyData);

    function loadPropertyData(){
      const number = propertyNumberInput.value.trim();
      const requestType = requestTypeSelect.value;

      if(!number){ statusDiv.innerText=""; return; }

      statusDiv.innerText="🔎 جاري التحقق...";
      statusDiv.className="status";

      google.script.run.withSuccessHandler(res=>{
        if(res && res.exists){
          if(requestType === "جديد"){
            statusDiv.innerText="⚠️ رقم العقار موجود مسبقًا، اختر تحديث بدلًا من جديد";
            statusDiv.className="status error";
            form.reset();
            document.querySelectorAll(".price-field").forEach(f=>f.style.display="none");
          } else if(requestType === "تحديث"){
            const data=res.data;
            document.querySelector("input[name='propertyName']").value=data.propertyName||"";
            document.querySelector("input[name='phoneNumber']").value=data.phoneNumber||"";
            document.querySelector("input[name='location']").value=data.location||"";
            document.querySelector("select[name='region']").value=data.region||"";
            document.querySelector("select[name='direction']").value=data.direction||"";
            document.querySelector("select[name='type']").value=data.type||"";
            document.querySelectorAll("input[name='facilities']").forEach(cb=>{
              cb.checked = data.facilities?.includes(cb.value) || false;
            });

            // تعبئة الأسعار وحذف الحقول غير المحددة
            ["offerPrice","midweekPrice","weekendPrice","overnightPrice","holidayPrice"].forEach(field=>{
              const cb = document.querySelector(`.checkbox-group input[value='${field}']`);
              const inputField = document.querySelector(`input[name='${field}']`);
              if(data[field]){
                cb.checked = true;
                inputField.value = data[field];
              } else {
                cb.checked = false;
                inputField.value = "";
              }
              togglePriceField(cb); // ← يضمن مسح القيمة القديمة وإخفاء الحقل إذا لم يكن محدد
            });

            statusDiv.innerText="✅ تم تحميل البيانات بنجاح";
            statusDiv.className="status";
          }
        } else {
          if(requestType==="جديد"){
            statusDiv.innerText="✅ رقم العقار جديد، يمكنك إدخال البيانات";
            statusDiv.className="status";
          } else {
            statusDiv.innerText="ℹ️ لم يتم العثور على بيانات لهذا الرقم";
            statusDiv.className="status";
            form.reset();
            document.querySelectorAll(".price-field").forEach(f=>f.style.display="none");
          }
        }
      }).getPropertyData(number);
    }

    form.addEventListener("submit", async function(e){
      e.preventDefault();
      const formData = new FormData(form);
      const submitButton = form.querySelector("button[type='submit']");
      submitButton.disabled = true;
      statusDiv.innerText = "⏳ جاري المعالجة...";
      statusDiv.className = "status";

      try {
        const selectedFacilities = Array.from(form.querySelectorAll("input[name='facilities']:checked")).map(cb => cb.value);

       function getPriceValue(fieldName){
  const cb = document.querySelector(`.checkbox-group input[value='${fieldName}']`);
  const inputField = document.querySelector(`input[name='${fieldName}']`);
  // إذا مفعل → ترسل القيمة الموجودة، إذا مش مفعل → ترسل 0
  return cb.checked ? inputField.value : 0;
}



        const output = {
          requestType: formData.get("requestType"),
          propertyNumber: formData.get("propertyNumber"),
          propertyName: formData.get("propertyName"),
          region: formData.get("region"),
          location: formData.get("location"),
          direction: formData.get("direction"),
          type: formData.get("type"),
          phoneNumber: formData.get("phoneNumber"),
          facilities: selectedFacilities,
          offerPrice: getPriceValue("offerPrice"),
          midweekPrice: getPriceValue("midweekPrice"),
          weekendPrice: getPriceValue("weekendPrice"),
          overnightPrice: getPriceValue("overnightPrice"),
          holidayPrice: getPriceValue("holidayPrice"),
          myFiles: []
        };

        const files = form.querySelector("input[name='myFiles']").files;
        for(let file of files){
          const reader = new FileReader();
          output.myFiles.push(await new Promise(resolve=>{
            reader.onload = () => resolve({name:file.name, type:file.type, data:reader.result.split(",")[1]});
            reader.readAsDataURL(file);
          }));
        }

        google.script.run.withSuccessHandler(res=>{
          if(res.status === "success"){
            statusDiv.innerHTML = `✅ تم الحفظ بنجاح<br><a href="${res.folderUrl}" target="_blank">📂 فتح مجلد الصور</a>`;
            form.reset();
            document.querySelectorAll(".price-field").forEach(f => f.style.display = "none");
          } else {
            statusDiv.innerText = "❌ " + res.message;
            statusDiv.className = "status error";
          }
          submitButton.disabled = false;
        }).uploadFiles(output);

      } catch(err){
        statusDiv.innerText = "⚠️ خطأ: " + err.message;
        statusDiv.className = "status error";
        submitButton.disabled = false;
      }
    });

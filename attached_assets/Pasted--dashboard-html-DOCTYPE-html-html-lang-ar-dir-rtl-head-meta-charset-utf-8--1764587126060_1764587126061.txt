/* ====== إعدادات ====== */
const API_URL = "https://script.google.com/macros/s/AKfycbwPeZKGJVdLRW3hGega8dOqtSMfQzJlNg7DWSSPcWiv3ko7JH220bE3FUTBeyUbbe5r/exec";

/* حالة التخزين */
let raw = [];
let charts = {}; // نخزن مثيل Chart.js هنا

/* ====== عناصر DOM ====== */
const totalRequestsEl = document.getElementById('totalRequests');
const totalUnitsEl = document.getElementById('totalUnits');

const growthEl = document.getElementById('growthPerc');

const filterDay = document.getElementById('filterDay');
const filterMonth = document.getElementById('filterMonth');
const filterYear = document.getElementById('filterYear');

const tabs = document.querySelectorAll('.tab');
tabs.forEach(t => t.addEventListener('click', () => {
  document.querySelectorAll('.tab').forEach(x=>x.classList.remove('active'));

/* ====== فتح تب ====== */
function openSection(id){
  document.getElementById('overview').style.display = id==='overview' ? 'block' : 'none';
  document.getElementById('charts').style.display = id==='charts' ? 'block' : 'none';
  document.getElementById('table').style.display = id==='table' ? 'block' : 'none';
}

/* ====== تحميل + عرض البيانات ====== */
async function loadAndRender(){
  try{
    const res = await fetch(API_URL);
    raw = await res.json();
    // normalize: ensure numeric fields
    raw = raw.map(r => ({
      id: r["رقم العقار"],
      name: (r["اسم العقار"]||'غير معروف').toString().trim(),
      orders: Number(r["عدد الطلبات"]) || 0,
      day: String(r["اليوم"]),
      month: String(r["الشهر"]),
      year: String(r["السنة"]),
      time: r["الوقت"]
    }));
    renderAll(raw);
    fillFilters(raw);
  }catch(err){
    console.error('Fetch error', err);
    alert('خطأ في جلب البيانات. تأكد من وصول رابط JSON.');
  }
}

/* ====== Fill Filters ====== */
function fillFilters(data){
  const days = Array.from(new Set(data.map(r=>r.day))).sort((a,b)=>a-b);
  const months = Array.from(new Set(data.map(r=>r.month))).sort((a,b)=>a-b);
  const years = Array.from(new Set(data.map(r=>r.year))).sort((a,b)=>a-b);

  filterDay.innerHTML = '<option value=\"\">اليوم</option>' + days.map(d=>`<option value="${d}">${d}</option>`).join('');
  filterMonth.innerHTML = '<option value=\"\">الشهر</option>' + months.map(m=>`<option value="${m}">${m}</option>`).join('');
  filterYear.innerHTML = '<option value=\"\">السنة</option>' + years.map(y=>`<option value="${y}">${y}</option>`).join('');
}

/* ====== Apply Filters ====== */
function applyFilters(){
  const fday = filterDay.value;
  const fmonth = filterMonth.value;
  const fyear = filterYear.value;
  const filtered = raw.filter(r=>{
    return (!fday || r.day===fday) && (!fmonth || r.month===fmonth) && (!fyear || r.year===fyear);
  });
  renderAll(filtered);
}

/* ====== Render All (cards, charts, table) ====== */
function renderAll(data){
  // جمع عدد الطلبات الحقيقي per unit
  const byUnit = {};
  const byMonth = {};
  data.forEach(r => {
    byUnit[r.name] = (byUnit[r.name]||0) + (Number(r.orders)||0);
    byMonth[r.month] = (byMonth[r.month]||0) + (Number(r.orders)||0);
  });

  // الكروت
  const totalOrders = Object.values(byUnit).reduce((s,x)=>s+x,0);
  totalRequestsEl.innerText = totalOrders;
  totalUnitsEl.innerText = Object.keys(byUnit).length;
  const sortedUnits = Object.entries(byUnit).sort((a,b)=>b[1]-a[1]);
  topUnitEl.innerText = sortedUnits[0] ? `${sortedUnits[0][0]} (${sortedUnits[0][1]})` : '-';

  // نمو حسب أشهر (نسبة نمو الشهر الأخير مقابل الذي قبله)
  const monthsSorted = Object.keys(byMonth).map(m=>Number(m)).sort((a,b)=>a-b).map(String);
  let growth = 0;
  if(monthsSorted.length >= 2){
    const last = byMonth[monthsSorted[monthsSorted.length-2]]||0;
    const now = byMonth[monthsSorted[monthsSorted.length-1]]||0;
    growth = last ? (((now - last) / last) * 100) : 100;
  }
  growthEl.innerText = `${growth===0?0:growth.toFixed(1)}%`;

  // جداول
  const tbody = document.querySelector('#dataTable tbody');
  tbody.innerHTML = '';
  // show each raw entry (not aggregated) so partner can inspect rows
  data.forEach(r => {
    const tr = document.createElement('tr');
    tr.innerHTML = `<td>A${r.id}</td><td>${r.name}</td><td>${r.orders}</td><td>${r.day}/${r.month}/${r.year}</td><td>${formatTime(r.time)}</td>`;
    tbody.appendChild(tr);
  });

  // مخططات (أعد استخدامها إن وجدت)
  // monthsChart
  const monthsLabels = Object.keys(byMonth).sort((a,b)=>a-b);
  const monthsValues = monthsLabels.map(k=>byMonth[k]||0);

  if(charts.monthsChart) charts.monthsChart.destroy();
  charts.monthsChart = new Chart(document.getElementById('monthsChart'), {
    type:'bar',
    data:{ labels: monthsLabels, datasets:[{ label:'طلبات', data: monthsValues, backgroundColor:'#2563eb' }] },
    options:{ responsive:true, plugins:{ legend:{display:false} } }
  });

  // topChart - top 10 units
  const top10 = sortedUnits.slice(0,10);
  if(charts.topChart) charts.topChart.destroy();
  charts.topChart = new Chart(document.getElementById('topChart'), {
    type:'bar',
    data:{ labels: top10.map(x=>x[0]), datasets:[{ label:'طلبات', data: top10.map(x=>x[1]), backgroundColor:'#06b6d4' }] },
    options:{ indexAxis: 'y', responsive:true, plugins:{ legend:{display:false} } }
  });

  // doughnut
  if(charts.doughnut) charts.doughnut.destroy();
  charts.doughnut = new Chart(document.getElementById('doughnutChart'), {
    type:'doughnut',
    data:{ labels: top10.map(x=>x[0]), datasets:[{ data: top10.map(x=>x[1]), backgroundColor: generateColors(top10.length) }] },
    options:{ responsive:true }
  });

  // line (trend)
  if(charts.line) charts.line.destroy();
  charts.line = new Chart(document.getElementById('lineChart'), {
    type:'line',
    data:{ labels: monthsLabels, datasets:[{ label:'ترند الطلب', data: monthsValues, borderColor:'#4c6ef5', fill:false, tension:0.3 }] },
    options:{ responsive:true }
  });
}

/* ====== Utilities ====== */
function generateColors(n){
  const palette = ['#06b6d4','#4c6ef5','#00b894','#ff7675','#ff9f1a','#845ef7','#2bcbba','#e76f51','#f9c74f','#90be6d'];
  return Array.from({length:n}).map((_,i)=>palette[i % palette.length]);
}
function formatTime(t){
  if(!t) return '';
  try{ const d = new Date(t); return d.toLocaleTimeString(); }catch(e){ return t; }
}

/* ====== Export CSV (Excel) ====== */
function exportCSV(){
  // Build aggregated CSV by unit (sum of عدد الطلبات)
  const byUnit = {};
  raw.forEach(r => byUnit[r.name] = (byUnit[r.name]||0) + Number(r.orders || 0));
  const rows = [['اسم العقار','عدد الطلبات']];
  Object.entries(byUnit).forEach(([k,v])=> rows.push([k,v]));

  const csv = rows.map(r => r.map(cell=>`"${(cell+'').replace(/"/g,'""')}"`).join(',')).join('\\n');
  const blob = Utilities.newBlob ? Utilities.newBlob(csv, 'text/csv') : new Blob([csv], { type: 'text/csv;charset=utf-8;' });
  // if Utilities not available (client-side) use download link:
  if(typeof Utilities === 'undefined'){
    const a = document.createElement('a');
    a.href = 'data:text/csv;charset=utf-8,' + encodeURIComponent(csv);
    a.download = 'requests_summary.csv';
    document.body.appendChild(a); a.click(); a.remove();
  }else{
    // this branch won't run in browser; kept for AppsScript server-side alternative
  }
}

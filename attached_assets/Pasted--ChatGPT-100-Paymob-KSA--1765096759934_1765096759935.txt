🛑 إنشاء Webhook Handler مع HMAC Verification

إذا لم تتحقق من HMAC، ستأتي بيانات الدفع ناقصة أو غير صحيحة.

✔ 2) دالة التحقق من HMAC
import crypto from "crypto";

function verifyPaymobHmac(body, hmac) {
  const secret = process.env.PAYMOB_HMAC;

  const ordered = {
    amount_cents: body.obj.amount_cents,
    created_at: body.obj.created_at,
    currency: body.obj.currency,
    error_occured: body.obj.error_occured,
    has_parent_transaction: body.obj.has_parent_transaction,
    id: body.obj.id,
    integration_id: body.obj.integration_id,
    is_3d_secure: body.obj.is_3d_secure,
    is_auth: body.obj.is_auth,
    is_capture: body.obj.is_capture,
    is_refunded: body.obj.is_refunded,
    is_standalone_payment: body.obj.is_standalone_payment,
    is_voided: body.obj.is_voided,
    order_id: body.obj.order_id,
    owner: body.obj.owner,
    pending: body.obj.pending,
    source_data_pan: body.obj.source_data.pan,
    source_data_sub_type: body.obj.source_data.sub_type,
    source_data_type: body.obj.source_data.type,
    success: body.obj.success
  };

  const query = Object.values(ordered).join("");

  const hash = crypto
    .createHmac("sha512", secret)
    .update(query)
    .digest("hex");

  return hash === hmac;
}

<script>
/* كود جافاسكريبت يبقى كما هو بدون تغيير الوظائف */
const propertyNumberInput = document.getElementById('propertyNumber');
const subscriptionInfo = document.getElementById('subscriptionInfo');
const nameSpan = document.getElementById('name');
const phoneSpan = document.getElementById('phone');
const subscriptionSpan = document.getElementById('subscriptionType');
const startSpan = document.getElementById('startDate');
const endSpan = document.getElementById('endDate');
const remainingSpan = document.getElementById('remainingDays');
const statusSub = document.getElementById('statusSub');
const subscriptionPriceSpan = document.getElementById('subscriptionPrice');
const receiptLink = document.getElementById('receiptLink');
const packagesDiv = document.getElementById('packages');
const receiptFilePackage = document.getElementById('receiptFilePackage');
const activatePackageBtn = document.getElementById('activatePackageBtn');
const statusDiv = document.getElementById('status');
const paymentPreview = document.getElementById('paymentPreview');
const previewName = document.getElementById('previewName');
const previewPhone = document.getElementById('previewPhone');
const previewSubscription = document.getElementById('previewSubscription');
const previewDays = document.getElementById('previewDays');
const previewPrice = document.getElementById('previewPrice');
const payBtn = document.getElementById('payBtn');
const refreshBtn = document.getElementById('refreshBtn');
let selectedPackage = null;

propertyNumberInput.addEventListener('input', ()=> {
  const val = propertyNumberInput.value.trim();
  if(/^\d{5}$/.test(val)){
    statusDiv.innerText = "جاري جلب البيانات...";
    google.script.run.withSuccessHandler(data=>{
      if(!data){
        statusDiv.innerText="❌ رقم العقار غير موجود"; 
        subscriptionInfo.style.display="none";
        packagesDiv.innerHTML="";
        activatePackageBtn.style.display="none";
        packageUpload.style.display="none";
        paymentPreview.style.display="none";
        payBtn.style.display="none";
        return;
      }

      subscriptionInfo.style.display="block";
      nameSpan.innerText = data.name;
      phoneSpan.innerText = data.phone;
      subscriptionSpan.innerText = data.subscriptionType;
      startSpan.innerText = data.startDate;
      endSpan.innerText = data.endDate;
      remainingSpan.innerText = data.remainingDays;
      subscriptionPriceSpan.innerText = data.subscriptionPrice || "-";
      receiptLink.href = data.receiptLink || "#";

      const remainingDays = parseInt(data.remainingDays);

      if(remainingDays <= 0){
        statusSub.innerText = "منتهي ❌";
        statusSub.className = "status-expired";
      } else {
        statusSub.innerText = "مفعل ✅";
        statusSub.className = "status-active";
      }

      // جلب الباقات
      google.script.run.withSuccessHandler(packages => {
        packagesDiv.innerHTML = "";
        if(packages.length > 0){
          packages.forEach(p => {
            const card = document.createElement('div');
            card.className = "package-card";
            card.innerHTML = `<h3>${p.name}</h3>
                              <p>المدة: ${p.days} يوم</p>
                              <p>الرسوم: ${p.price} ريال</p>`;
            card.addEventListener('click', ()=>{
              document.querySelectorAll('.package-card').forEach(c=>c.classList.remove('package-selected'));
              card.classList.add('package-selected');
              selectedPackage = p;

              activatePackageBtn.style.display = "block";
              packageUpload.style.display = "block";
              payBtn.style.display = "inline-block";

              // معاينة الدفع
              previewName.innerText = nameSpan.innerText;
              previewPhone.innerText = phoneSpan.innerText;
              previewSubscription.innerText = selectedPackage.name;
              previewDays.innerText = selectedPackage.days;
              previewPrice.innerText = selectedPackage.price;
              paymentPreview.style.display = "block";
            });
            packagesDiv.appendChild(card);
          });
          activatePackageBtn.style.display = "none";
          packageUpload.style.display = "none";
          payBtn.style.display = "none";
        } else {
          packagesDiv.innerHTML = "<p style='text-align:center;color:#999;'>لا توجد باقات متاحة حالياً</p>";
          activatePackageBtn.style.display = "none";
          packageUpload.style.display = "none";
          payBtn.style.display = "none";
        }

        if(remainingDays > 10){
          packagesDiv.innerHTML = "<p style='text-align:center; color:green; font-weight:bold;'>اشتراكك مفعل ولا يمكن تفعيل باقات إضافية في الوقت الحالي ✅</p>";
          activatePackageBtn.style.display = "none";
          packageUpload.style.display = "none";
          payBtn.style.display = "none";
        }

      }).getPackagesFromSheet();

      statusDiv.innerText="";
    }).getSubscriptionByProperty(val);
  } else {
    subscriptionInfo.style.display="none";
    packagesDiv.innerHTML="";
    activatePackageBtn.style.display="none";
    packageUpload.style.display="none";
    paymentPreview.style.display="none";
    payBtn.style.display="none";
    statusDiv.innerText="";
  }
});

let paymentInProgress = false;

payBtn.addEventListener('click', () => {
  if (!selectedPackage) {
    alert("الرجاء اختيار الباقة أولاً");
    return;
  }
  if (paymentInProgress) return;
  paymentInProgress = true;

  const subscriptionData = {
    name: nameSpan.innerText,
    phone: phoneSpan.innerText,
    propertyNumber: propertyNumberInput.value.trim(),
    subscriptionType: subscriptionSpan.innerText,
    startDate: startSpan.innerText,
    endDate: endSpan.innerText,
    remainingDays: remainingSpan.innerText,
    subscriptionPrice: subscriptionPriceSpan.innerText,
    packageName: selectedPackage.name,
    packagePrice: selectedPackage.price,
    timestamp: new Date().toISOString()
  };

  statusDiv.innerText = "⏳ جاري إنشاء رابط الدفع...";

  google.script.run.withSuccessHandler(checkoutUrl => {
    if (!checkoutUrl) {
      alert("❌ حدث خطأ أثناء إنشاء رابط الدفع");
      statusDiv.innerText = "";
      paymentInProgress = false;
      return;
    }

    const paymentWindow = window.open(checkoutUrl, "_blank");
    let remainingTime = 180; // 3 دقائق
    const CHECK_INTERVAL = 5000; // كل 5 ثواني

    // إخفاء الزر أولاً
    refreshBtn.style.display = "none";

    // إظهار الزر بعد 30 ثانية
    setTimeout(() => {
      refreshBtn.style.display = "inline-block";
    }, 30000);

    const interval = setInterval(() => {
      remainingTime -= CHECK_INTERVAL / 1000;
      if (remainingTime <= 0) {
        statusDiv.innerText = "❌ فشلت عملية الدفع (انتهت المهلة)";
        clearInterval(interval);
        paymentInProgress = false;
        refreshBtn.style.display = "none";
        if (!paymentWindow.closed) paymentWindow.close();
        return;
      }
      statusDiv.innerText = `⏳ جاري التحقق من حالة الدفع... (${Math.ceil(remainingTime)}s)`;
    }, CHECK_INTERVAL);

    // الزر لتحديث بيانات الاشتراك مباشرة من Sheet
    refreshBtn.onclick = () => {
      google.script.run.withSuccessHandler(res => {
        if (!res) {
          statusDiv.innerText = "❌ لم يتم العثور على بيانات الاشتراك";
          return;
        }

        // 👇 البيانات المهمة للتطابق مع ورقة المدفوعات
  const subscriptionData = {
  name: nameSpan.innerText,
  phone: phoneSpan.innerText,
  propertyNumber: propertyNumberInput.value.trim(),
  subscriptionType: subscriptionSpan.innerText,
  startDate: startSpan.innerText,
  endDate: endSpan.innerText,
  remainingDays: remainingSpan.innerText,
  subscriptionPrice: subscriptionPriceSpan.innerText,
  timestamp: new Date().toISOString()  // ← هنا المشكلة
};

        if (res.status === "مفعل") {
          clearInterval(interval);
          paymentInProgress = false;
          refreshBtn.style.display = "none";
          if (!paymentWindow.closed) paymentWindow.close();
        }
      
     // إخفاء الرسالة بعد 10 ثواني مثلاً بدلًا من الإخفاء الفوري
    setTimeout(() => {
      statusDiv.innerText = "";
    }, 10000);
  }).getSubscriptionByProperty(subscriptionData.propertyNumber);
};

  }).createPaymobCheckout(subscriptionData, selectedPackage);
});


// تفعيل الباقة مع رفع الإيصال
activatePackageBtn.addEventListener('click', ()=>{
  if(!selectedPackage){
    alert("الرجاء اختيار الباقة أولاً");
    return;
  }

  const file = receiptFilePackage.files[0];
  if(!file){
    alert("⚠️ رفع الإيصال إجباري لتفعيل الباقة");
    return;
  }

  statusDiv.innerText = "⏳ جاري تفعيل الباقة...";
  const reader = new FileReader();
  reader.onload = function(e){
    const payload = {
      propertyNumber: propertyNumberInput.value.trim(),
      packageName: selectedPackage.name,
      packageDays: selectedPackage.days,
      packagePrice: selectedPackage.price,
      receiptFile: {name:file.name, type:file.type, data:e.target.result.split(',')[1]}
    };
    google.script.run.withSuccessHandler(res=>{
      statusDiv.innerText = res.message;
      receiptFilePackage.value = "";
      activatePackageBtn.style.display = "none";
      packageUpload.style.display = "none";
      propertyNumberInput.dispatchEvent(new Event('input'));
    }).activateSubscriptionWithReceipt(payload);
  }
  reader.readAsDataURL(file);
});
</script>

function renderCardsWithProgress(data){
  filteredData = data;
  const container = document.getElementById('cards-container');
  container.innerHTML='';

  const trustedItems = data.filter(item => (item['اسم العقار']||'').trim() !== '');
  const normalItems = data.filter(item => (item['اسم العقار']||'').trim() === '');

  const allItems = [...trustedItems, ...normalItems];
  let totalImages = allItems.length * 10; // تقريب 10 صور لكل بطاقة
  let loadedImages = 0;

  function updateProgress(){
    const progress = Math.min(100, (loadedImages / totalImages * 100));
    const progressBar = document.getElementById('progressBar');
    if(progressBar) progressBar.style.width = progress + '%';
  }

  function preloadImagesWithProgress(imageUrls){
    return Promise.all(imageUrls.map(url=>{
      return new Promise(resolve=>{
        const img = new Image();
        img.src = url;
        img.onload = img.onerror = ()=>{
          loadedImages++;
          updateProgress();
          resolve();
        };
      });
    }));
  }

  function renderItems(items){
    return Promise.all(items.map(item=>{
      return new Promise(resolve=>{
        const id = item['رقم العقار']||'';
        google.script.run.withSuccessHandler(async images=>{
          const limitedImages = images.slice(0,10);
          renderCardWithImages(item, limitedImages); // تعرض البطاقة فوراً
          await preloadImagesWithProgress(limitedImages);
          resolve();
        }).getCachedImages(id);
      });
    }));
  }

  // نعرض الموثوق أولاً ثم الباقي
  return renderItems(trustedItems).then(()=>renderItems(normalItems));
}

// 🔹 دالة تحميل الصور بدون تغيير
function preloadImages(imageUrls){
  return Promise.all(imageUrls.map(url=>{
    return new Promise(resolve=>{
      const img = new Image();
      img.src = url;
      img.onload = resolve;
      img.onerror = resolve; // نتجاهل أي خطأ
    });
  }));
}

function setupFilters(){
  const filterButtons = document.querySelectorAll('#container .column');

  filterButtons.forEach(btn => {
    btn.addEventListener('click', () => {
      filterButtons.forEach(b => b.classList.remove('active'));
      btn.classList.add('active');

      const text = btn.innerText.trim();
      document.querySelectorAll('.card').forEach(card => {
        let fac = card.dataset.facilities || '';
        // توحيد النصوص: إزالة مسافات، أحرف متغيرة، تحويل لكل الحروف الصغيرة
        let facNormalized = fac.toLowerCase().replace(/\s+/g,'')
                             .replace(/[أإآ]/g,'ا')
                             .replace(/[ة]/g,'ه')
                             .replace(/[ى]/g,'ي');

        if(text === 'الكل'){
          card.style.display = 'flex';
        } else if(text === 'مناسب للمبيت'){
          card.style.display = facNormalized.includes('نوم') ? 'flex' : 'none';
          } else if(text === 'بدون مسبح'){
          card.style.display = facNormalized.includes('بدونمسبح') ? 'flex' : 'none';
          } else if(text === 'مسبح'){
          card.style.display = facNormalized.includes('مسبح') &&!facNormalized.includes('بدونمسبح') ? 'flex' : 'none';
          
          
        } else if(text === 'مناسب للمناسبات'){
          card.style.display = (
            facNormalized.includes('مناسبه') ||
            facNormalized.includes('مناسبة') ||
            facNormalized.includes('مناسبات') ||
            facNormalized.includes('صاله') ||
            facNormalized.includes('صالة')
          ) ? 'flex' : 'none';
        } else if(text === 'ألعاب مائية'){
          card.style.display = (
            facNormalized.includes('العابمائيه') ||
            facNormalized.includes('العابمائي') ||
            facNormalized.includes('ألعابمائية') ||
            facNormalized.includes('مائيه')
          ) ? 'flex' : 'none';
          } else if(text === 'شاليهات الشتاء'){
          card.style.display = (
            facNormalized.includes('خيمه') ||
            facNormalized.includes('خيمة') ||
            facNormalized.includes('خيمهعربيه') ||
            facNormalized.includes('عربيه') ||
            facNormalized.includes('خيمهشتويه') ||
            facNormalized.includes('شتويه') ||
            facNormalized.includes('بيتشعر') ||
            facNormalized.includes('مخيم') ||
            facNormalized.includes('مشب')
          ) ? 'flex' : 'none';
        } else {
          card.style.display = facNormalized.includes(text.replace(/\s+/g,'')) ? 'flex' : 'none';
        }
      });

      // إعادة تعيين الفلاتر المتقدمة
      ['directionSelect','regionSelect','typeSelect','facilitySearch']
        .forEach(id => document.getElementById(id).value = '');
    });
  });

  ['directionSelect','regionSelect','typeSelect','facilitySearch']
    .forEach(id => document.getElementById(id).addEventListener('input', applyAdvancedFilters));

  document.getElementById('all').click();
}


function applyAdvancedFilters(){
  const dir = document.getElementById('directionSelect').value.trim().toLowerCase();
  const region = document.getElementById('regionSelect').value.trim().toLowerCase();
  const type = document.getElementById('typeSelect').value.trim().toLowerCase();
  const facilitySearch = document.getElementById('facilitySearch').value.trim().toLowerCase();

  // القيم الصحيحة للسلايدر
  const minPrice = parseFloat(document.getElementById('priceSliderMin').value) || 0;
  const maxPrice = parseFloat(document.getElementById('priceSliderMax').value) || Infinity;

  // تحديث التسميات فوق السلايدر
  document.getElementById('minPriceLabel').innerText = minPrice;
  document.getElementById('maxPriceLabel').innerText = maxPrice;

  document.querySelectorAll('.card').forEach(card => {
    const textDir = (card.dataset.direction || '').toLowerCase();
    const textReg = (card.dataset.region || '').toLowerCase();
    const textType = (card.dataset.type || '').toLowerCase();
    const facilities = (card.dataset.facilities || '').toLowerCase();
    const location = (card.dataset.location || '').toLowerCase();
    const name = (card.dataset.name || '').toLowerCase();

    const searchableFields = [
      normalizeArabic(name),
      normalizeArabic(textType),
      normalizeArabic(textReg),
      normalizeArabic(textDir),
      normalizeArabic(location),
      normalizeArabic(facilities)
    ];

    // استخراج السعر من div الأسعار أو dataset.price
    let price = parseFloat(card.dataset.price || card.querySelector('.prices-box div span:last-child')?.innerText.replace(/[^\d]/g,'') || 0);

    const matchDir = !dir || textDir === dir;
    const matchReg = !region || textReg === region;
    const matchType = !type || textType === type;
    const matchFacilityOrLocation = !facilitySearch || searchableFields.some(f => f.includes(facilitySearch));
    const matchPrice = price >= minPrice && price <= maxPrice;

    card.style.display = (matchDir && matchReg && matchType && matchFacilityOrLocation && matchPrice) ? 'flex' : 'none';
  });

  // إزالة تفعيل الفلاتر العلوية عند تحريك السلايدر
  document.querySelectorAll('#container .column').forEach(b => b.classList.remove('active'));
}

// ربط السلايدر بالدالة
document.getElementById('priceSliderMin').addEventListener('input', applyAdvancedFilters);

function normalizeArabic(text){
  return text.toLowerCase()
             .replace(/[أإآا]/g,'ا')
             .replace(/[ة]/g,'ه')
             .replace(/[ى]/g,'ي')
             .replace(/[ء]/g,'')
             .replace(/[\u064B-\u0652]/g,'')
             .trim();
}
